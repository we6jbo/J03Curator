cmake_minimum_required(VERSION 3.21)
project(J03Curator VERSION 0.6 LANGUAGES CXX)

find_package(Qt6 REQUIRED COMPONENTS Core Sql)

install(TARGETS J03Curator RUNTIME DESTINATION bin)
